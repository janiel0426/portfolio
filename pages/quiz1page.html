Item Manager: WPF desktop app

A small Windows app that adds and removes items in a list, with warnings for empty input and for removing when nothing is selected.

Demo video

Shows adding items, removing a selected item, and the warning messages.

MainWindow.xaml.cs
using System.Windows;
using System.Windows.Media;

namespace ItemManagerApp
{
    public partial class MainWindow : Window
    {
        private const string PlaceholderText = "Enter new item...";

        public MainWindow()
        {
            InitializeComponent();
            SetPlaceholder(); // Set placeholder initially
        }

        private void AddItem_Click(object sender, RoutedEventArgs e)
        {
            string newItem = ItemTextBox.Text.Trim();

            if (string.IsNullOrEmpty(newItem) || newItem == PlaceholderText)
            {
                MessageBox.Show("Please enter a valid item.", "Input Error",
                    MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            ItemListBox.Items.Add(newItem);
            SetPlaceholder();
        }

        private void RemoveItem_Click(object sender, RoutedEventArgs e)
        {
            if (ItemListBox.SelectedItem == null)
            {
                MessageBox.Show("Please select an item to remove.", "Selection Error",
                    MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            ItemListBox.Items.Remove(ItemListBox.SelectedItem);
        }

        private void ItemTextBox_GotFocus(object sender, RoutedEventArgs e)
        {
            if (ItemTextBox.Text == PlaceholderText)
            {
                ItemTextBox.Text = "";
                ItemTextBox.Foreground = Brushes.Black;
            }
        }

        private void ItemTextBox_LostFocus(object sender, RoutedEventArgs e)
        {
            if (string.IsNullOrWhiteSpace(ItemTextBox.Text))
            {
                SetPlaceholder();
            }
        }

        private void SetPlaceholder()
        {
            ItemTextBox.Text = PlaceholderText;
            ItemTextBox.Foreground = Brushes.Gray;
        }
    }
}

What I did

  • Built the window layout in XAML using Grid, StackPanel, TextBox, ListBox and Buttons.
  • Wrote the C# click handlers that add and remove items.
  • Added validation with warning message boxes for empty input and for nothing selected.
  • Added placeholder text using the GotFocus and LostFocus events.
