script.js
/**
 * CEBU EASTERN COLLEGE - CLIENT-SIDE CONTROLLER
 * Handles navigation, mobile drawer, modal interaction, and simulated authentication.
 */

document.addEventListener("DOMContentLoaded", () => {
  // Demo Authentication Key
  const AUTH_KEY = "cec_session_active";
  const USER_KEY = "cec_user_name";

  // Elements
  const body = document.getElementById("mainBody") || document.body;
  const loginModal = document.getElementById("loginModal");
  const loginNavBtn = document.getElementById("loginNavBtn");
  const logoutNavBtn = document.getElementById("logoutNavBtn");
  const heroPortalBtn = document.getElementById("heroPortalBtn");
  const userGreeting = document.getElementById("userGreeting");
  const modalCloseBtns = document.querySelectorAll(".js-close-modal");
  const modalForm = document.getElementById("modalLoginForm");
  const modalAlert = document.getElementById("loginAlert");
  const standaloneForm = document.getElementById("standaloneLoginForm");
  const dashLogoutBtn = document.getElementById("dashLogoutBtn");
  const mobileMenuBtn = document.getElementById("mobileMenuBtn");
  const navMenu = document.getElementById("navMenu");

  // --- 1. STATE INITIALIZATION ---
  function updateAuthState() {
    const isLoggedIn = sessionStorage.getItem(AUTH_KEY) === "true";
    const username = sessionStorage.getItem(USER_KEY) || "Admin";

    if (loginNavBtn) {
      if (isLoggedIn) {
        loginNavBtn.classList.add("is-hidden");
        if (logoutNavBtn) logoutNavBtn.classList.add("is-active");
        if (userGreeting) {
          userGreeting.textContent = `Welcome, ${username}`;
          userGreeting.classList.add("is-active");
        }
      } else {
        loginNavBtn.classList.remove("is-hidden");
        if (logoutNavBtn) logoutNavBtn.classList.remove("is-active");
        if (userGreeting) userGreeting.classList.remove("is-active");
      }
    }

    // Dashboard Route Protection
    if (window.location.pathname.endsWith("dashboard.html")) {
      if (!isLoggedIn) {
        window.location.replace("login.html");
      } else {
        const welcomeSpan = document.getElementById("dashWelcomeUser");
        const nameSpan = document.getElementById("dashStudentName");
        if (welcomeSpan) welcomeSpan.textContent = username;
        if (nameSpan) nameSpan.textContent = `${username} (Student)`;
      }
    }
  }

  // --- 2. MODAL CONTROLS (As in Demo Video) ---
  function openModal() {
    if (!loginModal) return;
    loginModal.classList.add("is-active");
    body.classList.add("is-fixed");
    if (modalAlert) modalAlert.style.display = "none";
    const userField = document.getElementById("modalUsername");
    if (userField) userField.focus();
  }

  function closeModal() {
    if (!loginModal) return;
    loginModal.classList.remove("is-active");
    body.classList.remove("is-fixed");
  }

  if (loginNavBtn) loginNavBtn.addEventListener("click", openModal);
  if (heroPortalBtn) heroPortalBtn.addEventListener("click", openModal);

  modalCloseBtns.forEach(btn => {
    btn.addEventListener("click", closeModal);
  });

  if (loginModal) {
    loginModal.addEventListener("click", (e) => {
      if (e.target === loginModal) closeModal();
    });
  }

  // Escape key closes modal
  document.addEventListener("keydown", (e) => {
    if (e.key === "Escape" && loginModal && loginModal.classList.contains("is-active")) {
      closeModal();
    }
  });

  // --- 3. DEMO AUTHENTICATION LOGIC ---
  // Video credentials: username "admin", password "123"
  function validateDemoLogin(username, password) {
    const cleanUser = username.trim();
    const cleanPass = password.trim();
    return cleanUser === "admin" && cleanPass === "123";
  }

  // A. Modal Form Submission
  if (modalForm) {
    modalForm.addEventListener("submit", (e) => {
      e.preventDefault();
      const user = document.getElementById("modalUsername").value;
      const pass = document.getElementById("modalPassword").value;

      if (validateDemoLogin(user, pass)) {
        sessionStorage.setItem(AUTH_KEY, "true");
        sessionStorage.setItem(USER_KEY, user);
        closeModal();
        updateAuthState();
        // Option: Redirect to dashboard or stay with "Welcome Admin" banner
        const redirectToDash = confirm("Login successful! Would you like to enter your Student Dashboard?");
        if (redirectToDash) {
          window.location.href = "dashboard.html";
        }
      } else {
        if (modalAlert) {
          modalAlert.textContent = "Incorrect username or password. (Hint: admin / 123)";
          modalAlert.style.display = "block";
        }
      }
    });
  }

  // B. Standalone Login Page Submission
  if (standaloneForm) {
    standaloneForm.addEventListener("submit", (e) => {
      e.preventDefault();
      const user = document.getElementById("pageUser").value;
      const pass = document.getElementById("pagePass").value;
      const alertBox = document.getElementById("pageLoginAlert");

      if (validateDemoLogin(user, pass)) {
        sessionStorage.setItem(AUTH_KEY, "true");
        sessionStorage.setItem(USER_KEY, user);
        window.location.href = "dashboard.html";
      } else {
        if (alertBox) {
          alertBox.textContent = "Invalid credentials. Use demo: admin / 123";
          alertBox.style.display = "block";
        }
      }
    });
  }

  // C. Logout Actions
  function performLogout() {
    sessionStorage.removeItem(AUTH_KEY);
    sessionStorage.removeItem(USER_KEY);
    updateAuthState();
    if (window.location.pathname.endsWith("dashboard.html")) {
      window.location.href = "index.html";
    }
  }

  if (logoutNavBtn) logoutNavBtn.addEventListener("click", performLogout);
  if (dashLogoutBtn) dashLogoutBtn.addEventListener("click", performLogout);

  // --- 4. RESPONSIVE MOBILE NAVIGATION ---
  if (mobileMenuBtn && navMenu) {
    mobileMenuBtn.addEventListener("click", () => {
      const isExpanded = mobileMenuBtn.getAttribute("aria-expanded") === "true";
      mobileMenuBtn.setAttribute("aria-expanded", !isExpanded);
      navMenu.classList.toggle("is-open");
    });

    // Close menu when clicking link
    navMenu.querySelectorAll(".nav-link").forEach(link => {
      link.addEventListener("click", () => {
        navMenu.classList.remove("is-open");
        mobileMenuBtn.setAttribute("aria-expanded", "false");
      });
    });
  }

  // Run initial state update
  updateAuthState();
});
